# Token and cost check, recovered from the local Claude Code session logs

Date: 2026-10-02 (run after S02, about 23:20)
Why: the top block of /usage (Total cost, Usage by model) was not saved at the end of S01. This file recovers S01's tokens from the session log
and checks the method against S02, whose /usage figures were saved.

Method
- Claude Code keeps one log file per session under the user's .claude\projects folder, named <session id>.jsonl.
- The script below reads one log, keeps the last usage entry for each message (message id + request id), and adds up the token fields by model.
- It only reads. It prints no file paths. It does not see Haiku background calls.
- Session ids: S01 = 03d0fa39-daf5-4873-a4a3-737ef9787d46, S02 = f30c058e-bf4a-4b68-bf62-f1369dd3e161.

Script (PowerShell)

function Sum-Usage([string]$id) {
  $f = Get-ChildItem "$env:USERPROFILE\.claude\projects" -Recurse -Filter "$id.jsonl" | Select-Object -First 1
  $seen = @{}
  foreach ($line in [System.IO.File]::ReadLines($f.FullName)) {
    if ($line -notlike '*"usage"*') { continue }
    try { $o = $line | ConvertFrom-Json } catch { continue }
    $m = $o.message
    if (-not $m -or -not $m.usage) { continue }
    $seen["$($m.id)|$($o.requestId)"] = [pscustomobject]@{ model=$m.model; i=[long]$m.usage.input_tokens; o=[long]$m.usage.output_tokens; cr=[long]$m.usage.cache_read_input_tokens; cw=[long]$m.usage.cache_creation_input_tokens }
  }
  $seen.Values | Group-Object model | ForEach-Object { [pscustomobject]@{ session=$id.Substring(0,8); model=$_.Name; messages=$_.Count; input=($_.Group|Measure-Object i -Sum).Sum; output=($_.Group|Measure-Object o -Sum).Sum; cache_read=($_.Group|Measure-Object cr -Sum).Sum; cache_write=($_.Group|Measure-Object cw -Sum).Sum } }
}
