## Source: compare-bookfinder.js
// Line-by-line comparison of the original BookFinder.js and BookFinder.ts.
// Every removed original line must be explained by an added TS line which, after stripping ONLY the
// spec-listed TypeScript syntax, is byte-identical to it. Every leftover added line must match a listed
// pure addition. Everything else is reported as UNLISTED.
// Usage: node compare-bookfinder.js <original.js> <BookFinder.ts>
const fs = require('fs')
const { execFileSync } = require('child_process')

const [origPath, tsPath] = process.argv.slice(2)
let diffText = ''
try {
  diffText = execFileSync('git', ['diff', '--no-index', '--patience', '-U0', '--no-color', '--', origPath, tsPath], { encoding: 'utf8' })
} catch (e) {
  diffText = e.stdout // exit 1 = differences found
}

// ---- parse hunks ----
const hunks = []
let cur = null
for (const line of diffText.split('\n')) {
  const m = line.match(/^@@ -(\d+)(?:,(\d+))? \+(\d+)(?:,(\d+))? @@/)
  if (m) {
    cur = { oldStart: +m[1], newStart: +m[3], removed: [], added: [] }
    hunks.push(cur)
    continue
  }
  if (!cur || line.startsWith('---') || line.startsWith('+++')) continue
  if (line.startsWith('-')) cur.removed.push({ no: cur.oldStart + cur.removed.length, text: line.slice(1) })
  else if (line.startsWith('+')) cur.added.push({ no: cur.newStart + cur.added.length, text: line.slice(1) })
}

// ---- normalisation: strip only spec-listed TS syntax ----
const CASTS = [' as OpenLibraryIsbnLookupResult', ' as BookProviderResponse<OpenLibraryBookResult>', ' as BookProviderResponse<BookSearchResult>', ' as string']

function splitTopLevel(s) {
  const parts = []
  let depth = 0
  let start = 0
  for (let i = 0; i < s.length; i++) {
    const c = s[i]
    if ('(<[{'.includes(c)) depth++
    else if (')>]}'.includes(c)) depth--
    else if (c === ',' && depth === 0) {
      parts.push(s.slice(start, i))
      start = i + 1
    }
  }
  parts.push(s.slice(start))
  return parts
}

function stripParamTypes(line) {
  const open = line.indexOf('(')
  if (open < 0) return line
  let depth = 0
  let close = -1
  for (let i = open; i < line.length; i++) {
    if (line[i] === '(') depth++
    else if (line[i] === ')') {
      depth--
      if (depth === 0) {
        close = i
        break
      }
    }
  }
  if (close < 0) return line
  const inner = line.slice(open + 1, close)
  if (!inner.includes(':')) return line
  const params = splitTopLevel(inner).map((p) => {
    const m = p.match(/^(\s*#?\w+)\??: ([\s\S]*?)(\s=\s[\s\S]*)?$/)
    return m ? m[1] + (m[3] || '') : p
  })
  return line.slice(0, open + 1) + params.join(',') + line.slice(close)
}

function normalise(tsLine) {
  const applied = []
  let s = tsLine
  for (const c of CASTS) {
    if (s.includes(c)) {
      s = s.split(c).join('')
      applied.push('cast')
    }
  }
  if (/^\s*(async |static |get |function |#)?[\w#]*\s*\(.*\)(: [^{]+)? \{$/.test(s) || /^\s*const calculateTitleScore = \(/.test(s) || /^\s*constructor\(/.test(s)) {
    const before = s
    s = s.replace(/\): [^{]+ \{$/, ') {')
    s = stripParamTypes(s)
    if (s !== before) applied.push('signature')
  }
  const local = s.replace(/^(\s*(?:const|let|var) \w+): .+? = /, '$1 = ').replace(/^(\s*let \w+): [^=]+$/, '$1')
  if (local !== s) {
    s = local
    applied.push('local-annotation')
  }
  const num = s.replace(/Number\(((?:[^()]|\([^()]*\))*)\)/g, '$1')
  if (num !== s) {
    s = num
    applied.push('Number')
  }
  return { s, applied }
}

function specIdFor(origLine, applied) {
  const ids = []
  if (applied.includes('cast')) {
    if (origLine.includes('isbnLookup')) ids.push('§6.1 C1')
    else if (origLine.includes('searchTitle')) ids.push('§6.1 C2')
    else if (origLine.includes('googleBooks.search')) ids.push('§6.1 C3+C4')
    else if (origLine.includes('fantLab.search')) ids.push('§6.1 C5+C6')
    else if (origLine.includes('audible.search')) ids.push('§6.1 C7+C8+C9')
    else if (origLine.includes('customProviderAdapter.search')) ids.push('§6.1 C10+C11')
    else ids.push('cast NOT IN §6.1')
  }
  if (applied.includes('Number')) {
    if (origLine.includes('b.titleDistance >')) ids.push('§6.2 N1')
    else if (origLine.includes('b.authorDistance >')) ids.push('§6.2 N2')
    else if (origLine.includes('b.totalPossibleDistance < 5')) ids.push('§6.2 N3')
    else if (origLine.includes('a.totalDistance - b.totalDistance')) ids.push('§6.2 N4')
    else if (origLine.includes('onlyDigits.test(a) - onlyDigits.test(b)')) ids.push('§6.2 N5')
    else ids.push('Number() NOT IN §6.2')
  }
  if (applied.includes('signature')) ids.push(origLine.includes('calculateTitleScore') ? 'ledger ruling: calculateTitleScore param types' : '§5.2 signature')
  if (applied.includes('local-annotation')) ids.push('§5.2 local annotation')
  return ids.join(' + ')
}

// ---- pure-addition rules ----
const TS_TEXT = fs.readFileSync(tsPath, 'utf8').split('\n')
const classLine = TS_TEXT.findIndex((l) => l === 'class BookFinder {') + 1
function pureAdditionId(a) {
  const t = a.text
  if (/^import \w+ = require\('[^']+'\)$/.test(t)) return '§4 import = require (part of E1/E2 pair)'
  if (/^\s+declare #?\w+: .+$/.test(t)) return '§4 declare field'
  if (t === '  // isbn also accepts BookSearchOptions because findCovers() passes its options object in this slot (SP-S01 bug #1)') return '§6.4 K1'
  if (a.no < classLine) {
    if (t.startsWith('// lib.d.ts types isNaN(number)') || t === 'declare function isNaN(value: unknown): boolean') return '§6.3 D1'
    if (/^const \{ levenshteinDistance, levenshteinSimilarity, escapeRegExp, isValidASIN \}: \{$/.test(t) || /^  (levenshteinDistance|levenshteinSimilarity|escapeRegExp|isValidASIN): \(/.test(t) || t === '} = utils') return '§4 two-line utils import + U3 (approved deviation)'
    if (t === '') return 'blank line separating new top-level declarations (§6.3/§5.1)'
    return '§5.1 interface / type alias'
  }
  if (t === '') return 'blank line after declare fields (§4)'
  return null
}

// ---- classify ----
const rows = []
const unlisted = []
for (const h of hunks) {
  const added = h.added.map((a) => ({ ...a, used: false, norm: normalise(a.text) }))
  for (const r of h.removed) {
    let match = null
    if (r.text === "const { levenshteinDistance, levenshteinSimilarity, escapeRegExp, isValidASIN } = require('../utils/index')") {
      match = added.find((a) => !a.used && a.text === "import utils = require('../utils/index')")
      if (match) {
        match.used = true
        rows.push(`orig ${r.no} -> ts ${match.no}: §4 two-line utils import (E2) [+ type literal lines below, U3]`)
        continue
      }
    }
    if (r.text === 'module.exports = new BookFinder()') {
      match = added.find((a) => !a.used && a.text === 'export = new BookFinder()')
      if (match) {
        match.used = true
        rows.push(`orig ${r.no} -> ts ${match.no}: §4 export =`)
        continue
      }
    }
    const req = r.text.match(/^const (\w+) = require\('([^']+)'\)$/)
    if (req) {
      match = added.find((a) => !a.used && a.text === `import ${req[1]} = require('${req[2]}')`)
      if (match) {
        match.used = true
        rows.push(`orig ${r.no} -> ts ${match.no}: §4 import = require (E1)`)
        continue
      }
    }
    match = added.find((a) => !a.used && a.norm.s === r.text && a.norm.applied.length)
    if (match) {
      match.used = true
      rows.push(`orig ${r.no} -> ts ${match.no}: ${specIdFor(r.text, match.norm.applied)}`)
      continue
    }
    unlisted.push(`REMOVED/CHANGED orig ${r.no}: ${JSON.stringify(r.text)}`)
  }
  for (const a of added) {
    if (a.used) continue
    const id = pureAdditionId(a)
    if (id) rows.push(`(added) ts ${a.no}: ${id}`)
    else unlisted.push(`ADDED ts ${a.no}: ${JSON.stringify(a.text)}`)
  }
}
