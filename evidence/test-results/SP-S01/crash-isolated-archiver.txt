// Program containing ONLY the given file (plus what it imports); declaration emit to a no-op writer.
const ts = require('D:/SEM-2/574/Project/audiobookshelf/node_modules/typescript')
const root = 'D:/SEM-2/574/Project/audiobookshelf'
const cfg = ts.readConfigFile(root + '/tsconfig.server.json', ts.sys.readFile)
const parsed = ts.parseJsonConfigFileContent(cfg.config, ts.sys, root)
const options = { ...parsed.options, incremental: false, tsBuildInfoFile: undefined, noEmit: false, declaration: true, emitDeclarationOnly: true, noEmitOnError: false, outDir: process.argv[3] }

const program = ts.createProgram([file], options)
const inProgram = program.getSourceFiles().map((s) => s.fileName).filter((f) => !f.includes('/node_modules/'))
console.log('program files (non node_modules):', inProgram.length, '| any finder in program:', inProgram.some((f) => f.includes('/finders/')))
